import React, { useState, useEffect } from 'react';
import { 
  StyleSheet, Text, View, TextInput, TouchableOpacity, 
  FlatList, KeyboardAvoidingView, Platform, ScrollView, Alert
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { Trash2, Plus, ShoppingBag, History, CheckCircle2 } from 'lucide-react-native';

export default function App() {
  const [tab, setTab] = useState('current');
  const [name, setName] = useState('');
  const [price, setPrice] = useState('');
  const [list, setList] = useState([]);
  const [history, setHistory] = useState([]);

  useEffect(() => { loadData(); }, []);

  const loadData = async () => {
    const savedList = await AsyncStorage.getItem('@list_v4');
    const savedHistory = await AsyncStorage.getItem('@history_v4');
    if (savedList) setList(JSON.parse(savedList));
    if (savedHistory) setHistory(JSON.parse(savedHistory));
  };

  const saveData = async (newList, newHistory) => {
    setList(newList);
    setHistory(newHistory);
    await AsyncStorage.setItem('@list_v4', JSON.stringify(newList));
    await AsyncStorage.setItem('@history_v4', JSON.stringify(newHistory));
  };

  // Função para adicionar emoji automático baseado no nome
  const getEmoji = (text) => {
    const lowerText = text.toLowerCase();
    if (lowerText.includes('leite')) return '🥛';
    if (lowerText.includes('pão') || lowerText.includes('pao')) return '🍞';
    if (lowerText.includes('carne') || lowerText.includes('frango')) return '🥩';
    if (lowerText.includes('maçã') || lowerText.includes('maca') || lowerText.includes('fruta')) return '🍎';
    if (lowerText.includes('cerveja') || lowerText.includes('vinho')) return '🍺';
    if (lowerText.includes('limpeza') || lowerText.includes('sabão')) return '🧼';
    if (lowerText.includes('café')) return '☕';
    return '📦'; // Ícone padrão de pacote
  };

  const addItem = () => {
    if (!name.trim()) return;
    const newItem = { 
      id: Date.now().toString(), 
      text: name, 
      emoji: getEmoji(name),
      price: parseFloat(price.replace(',', '.')) || 0,
      bought: false 
    };
    saveData([...list, newItem], history);
    setName(''); setPrice('');
  };

  const finishPurchase = () => {
    if (list.length === 0) return;
    const total = list.reduce((acc, curr) => acc + curr.price, 0);
    const newEntry = { id: Date.now().toString(), date: new Date().toLocaleDateString('pt-BR'), total: total, itemsCount: list.length };
    Alert.alert("Compra Finalizada", `R$ ${total.toFixed(2)} guardado no histórico!`);
    saveData([], [newEntry, ...history]);
    setTab('history');
  };

  const totalCurrent = list.reduce((acc, curr) => acc + curr.price, 0);

  return (
    <View style={styles.container}>
      {/* Tab Navigation */}
      <View style={styles.tabBar}>
        <TouchableOpacity onPress={() => setTab('current')} style={[styles.tab, tab === 'current' && styles.activeTab]}>
          <Text style={[styles.tabText, tab === 'current' && styles.activeTabText]}>Lista</Text>
        </TouchableOpacity>
        <TouchableOpacity onPress={() => setTab('history')} style={[styles.tab, tab === 'history' && styles.activeTab]}>
          <Text style={[styles.tabText, tab === 'history' && styles.activeTabText]}>Histórico</Text>
        </TouchableOpacity>
      </View>

      {tab === 'current' ? (
        <>
          <View style={styles.bagHeader}>
            <View style={styles.bagTop} />
            <Text style={styles.title}>Minha Sacola</Text>
            <Text style={styles.bagTotal}>Total: R$ {totalCurrent.toFixed(2)}</Text>
          </View>

          <FlatList
            data={list}
            style={styles.list}
            renderItem={({ item }) => (
              <View style={styles.card}>
                <Text style={styles.emoji}>{item.emoji}</Text>
                <View style={{ flex: 1 }}>
                  <Text style={styles.itemText}>{item.text}</Text>
                  <Text style={styles.priceTag}>R$ {item.price.toFixed(2)}</Text>
                </View>
                <TouchableOpacity onPress={() => saveData(list.filter(i => i.id !== item.id), history)}>
                  <Trash2 color="#7F5539" size={18} />
                </TouchableOpacity>
              </View>
            )}
          />

          <TouchableOpacity style={styles.finishBtn} onPress={finishPurchase}>
            <CheckCircle2 color="white" size={20} />
            <Text style={styles.finishBtnText}>Finalizar Compra</Text>
          </TouchableOpacity>

          <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : 'height'} style={styles.inputArea}>
            <TextInput style={[styles.input, { flex: 2 }]} placeholder="Ex: Leite" value={name} onChangeText={setName} />
            <TextInput style={[styles.input, { flex: 1 }]} placeholder="R$" keyboardType="numeric" value={price} onChangeText={setPrice} />
            <TouchableOpacity style={styles.addBtn} onPress={addItem}><Plus color="white" size={24} /></TouchableOpacity>
          </KeyboardAvoidingView>
        </>
      ) : (
        <ScrollView style={styles.list}>
          <Text style={styles.title}>Histórico</Text>
          {history.map(item => (
            <View key={item.id} style={styles.historyCard}>
              <View>
                <Text style={styles.historyDate}>{item.date}</Text>
                <Text style={styles.historySub}>{item.itemsCount} produtos</Text>
              </View>
              <Text style={styles.historyTotal}>R$ {item.total.toFixed(2)}</Text>
            </View>
          ))}
        </ScrollView>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#F5F5DC', paddingHorizontal: 20, paddingTop: 50 },
  tabBar: { flexDirection: 'row', marginBottom: 20, backgroundColor: '#E6CCB2', borderRadius: 25, padding: 4 },
  tab: { flex: 1, paddingVertical: 10, alignItems: 'center', borderRadius: 20 },
  activeTab: { backgroundColor: '#7F5539' },
  tabText: { color: '#7F5539', fontWeight: 'bold' },
  activeTabText: { color: 'white' },
  bagHeader: { backgroundColor: '#E6CCB2', padding: 20, borderRadius: 15, borderTopWidth: 8, borderTopColor: '#B08968', marginBottom: 15 },
  title: { fontSize: 22, fontWeight: 'bold', color: '#432818' },
  bagTotal: { fontSize: 18, color: '#7F5539', fontWeight: '600' },
  list: { flex: 1 },
  card: { backgroundColor: '#EDE0D4', padding: 15, borderRadius: 12, flexDirection: 'row', alignItems: 'center', marginBottom: 10, borderBottomWidth: 2, borderBottomColor: '#DDB892' },
  emoji: { fontSize: 24, marginRight: 15 },
  itemText: { color: '#432818', fontSize: 16, fontWeight: 'bold' },
  priceTag: { color: '#9C6644', fontSize: 14 },
  inputArea: { flexDirection: 'row', gap: 10, paddingVertical: 20 },
  input: { backgroundColor: '#EDE0D4', padding: 12, borderRadius: 10, color: '#432818' },
  addBtn: { backgroundColor: '#7F5539', padding: 12, borderRadius: 10, justifyContent: 'center' },
  finishBtn: { backgroundColor: '#432818', flexDirection: 'row', padding: 15, borderRadius: 12, justifyContent: 'center', alignItems: 'center', gap: 10 },
  finishBtnText: { color: 'white', fontWeight: 'bold' },
  historyCard: { padding: 15, backgroundColor: '#E6CCB2', borderRadius: 12, flexDirection: 'row', justifyContent: 'space-between', marginBottom: 10 },
  historyDate: { fontWeight: 'bold', color: '#432818' },
  historyTotal: { color: '#7F5539', fontWeight: 'bold' }
});
